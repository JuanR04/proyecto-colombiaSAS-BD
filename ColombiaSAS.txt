--- Crear tabla Tipo_plan
CREATE TABLE Tipo_plan (
    id_tipoplan SERIAL PRIMARY KEY,
    nombre_plan VARCHAR(50) NOT NULL
);

-- Crear tabla planes
CREATE TABLE planes (
    id_plan SERIAL PRIMARY KEY,
    id_tipoplan INT NOT NULL,
    nombre_plan VARCHAR(100),
    minutos INT,
    gigas INT,
    mensajes INT,
    precio_plan_postpago DECIMAL(10, 2),
	precio_plan_prepago DECIMAL(10,2),
    valor_min_adicional DECIMAL(10, 2),
    descripcion TEXT,
    FOREIGN KEY (id_tipoplan) REFERENCES Tipo_plan (id_tipoplan)
);
create table consumiblesxcontrato(
	id_consumibles serial primary key,
	nombre_plan VARCHAR(100),
    minutos INT,
    gigas INT,
    mensajes INT,
    precio_plan_postpago DECIMAL(10, 2),
	precio_plan_prepago DECIMAL(10,2),
    valor_min_adicional DECIMAL(10, 2),
    descripcion TEXT
);
-- Crear tabla Saldo_prepago
CREATE TABLE Saldo_prepago (
    id_saldo SERIAL PRIMARY KEY,
    numero_abonado VARCHAR(20),
    cant_saldo DECIMAL(10, 2)
);

-- Crear tabla Empresas
CREATE TABLE Empresas (
    id_empresa SERIAL PRIMARY KEY,
    nombre VARCHAR(100),
    ciudad VARCHAR(50)
);

-- Crear tabla Clientes
CREATE TABLE Clientes (
    id_cliente SERIAL PRIMARY KEY,
    tipo_documento VARCHAR(20),
    nombre VARCHAR(50),
    apellido VARCHAR(50),
    direccion VARCHAR(100),
    fecha_nacimiento DATE,
    sexo VARCHAR(10),
    telefono VARCHAR(20),
    ocupacion VARCHAR(50),
    rango_salarial VARCHAR(50)
);

-- Crear tabla Contratos
CREATE TABLE Contratos (
    id_contrato SERIAL PRIMARY KEY,
    id_plan INT,
    id_cliente INT,
    id_saldo INT,
    id_empresa INT,
	id_consumibles int,
    duracion INT,
	fecha_creacion date,
	FOREIGN KEY (id_consumibles) references consumiblesxcontrato(id_consumibles), 
    FOREIGN KEY (id_plan) REFERENCES planes (id_plan),
    FOREIGN KEY (id_cliente) REFERENCES Clientes (id_cliente),
    FOREIGN KEY (id_saldo) REFERENCES Saldo_prepago (id_saldo),
    FOREIGN KEY (id_empresa) REFERENCES Empresas (id_empresa)
);
-- Crear tabla Facturacion
CREATE TABLE Facturacion (
    id_detalle SERIAL PRIMARY KEY,
    num_factura INT,
    id_contrato INT,
    minutos_consumidos INT,
    precio_unitario_minuto DECIMAL(10, 2),
    consumo_plan DECIMAL(10, 2),
    valor_pagar DECIMAL(10, 2),
    impuestos DECIMAL(10, 2),
    fecha_inicio DATE,
    fecha_final DATE,
    FOREIGN KEY (id_contrato) REFERENCES Contratos (id_contrato)
);
CREATE TABLE Recargas (
    id_recargas SERIAL PRIMARY KEY,
	id_contrato int references Contratos (id_contrato),
    monto DECIMAL(10, 2),
    fecha DATE
);
-- Crear tabla Llamadas
CREATE TABLE Llamadas (
    id_llamada SERIAL PRIMARY KEY,
	id_contrato int REFERENCES contratos(id_contrato),
    numero_abonado VARCHAR(20),
    numero_tel_destino VARCHAR(20),
    minutos_consumidos INT,
    fecha DATE,
    ciudad_del_abonado VARCHAR(50),
    ciudad_destino VARCHAR(50),
    operador_destino VARCHAR(50)
);

CREATE OR REPLACE FUNCTION consumir_saldo_llamada()
RETURNS TRIGGER AS $$
DECLARE
    v_plan_id INT;
    v_minutos_restantes INT;
    v_minutos_a_consumir INT;
    v_saldo_restante DECIMAL(10, 2);
    v_valor_min_adicional DECIMAL(10, 2);
BEGIN
    -- Obtener el ID del plan del contrato
    SELECT id_plan INTO v_plan_id
    FROM Contratos
    WHERE id_contrato = NEW.id_contrato;

    -- Obtener los minutos restantes y el valor adicional por minuto del plan
    SELECT minutos, valor_min_adicional INTO v_minutos_restantes, v_valor_min_adicional
    FROM consumiblesxcontrato
    WHERE id_consumibles = (
        SELECT id_consumibles
        FROM Contratos
        WHERE id_contrato = NEW.id_contrato
    );

    -- Determinar si el cliente tiene minutos en el plan
    IF v_minutos_restantes >= NEW.minutos_consumidos THEN
        -- Consumir minutos del plan
        UPDATE consumiblesxcontrato
        SET minutos = minutos - NEW.minutos_consumidos
        WHERE id_consumibles = (
            SELECT id_consumibles
            FROM Contratos
            WHERE id_contrato = NEW.id_contrato
        );
    ELSE
        -- Consumir los minutos restantes del plan
        v_minutos_a_consumir := NEW.minutos_consumidos - v_minutos_restantes;
        UPDATE consumiblesxcontrato
        SET minutos = 0
        WHERE id_consumibles = (
            SELECT id_consumibles
            FROM Contratos
            WHERE id_contrato = NEW.id_contrato
        );

        -- Consumir el saldo restante del cliente
        SELECT cant_saldo INTO v_saldo_restante
        FROM Saldo_prepago
        WHERE numero_abonado = (
            SELECT telefono
            FROM Clientes
            WHERE id_cliente = (
                SELECT id_cliente
                FROM Contratos
                WHERE id_contrato = NEW.id_contrato
            )
        );

        IF v_saldo_restante >= (v_minutos_a_consumir * v_valor_min_adicional) THEN
            UPDATE Saldo_prepago
            SET cant_saldo = cant_saldo - (v_minutos_a_consumir * v_valor_min_adicional)
            WHERE numero_abonado = (
                SELECT telefono
                FROM Clientes
                WHERE id_cliente = (
                    SELECT id_cliente
                    FROM Contratos
                    WHERE id_contrato = NEW.id_contrato
                )
            );
        ELSE
            RAISE EXCEPTION 'Saldo insuficiente para realizar la llamada';
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_consumir_saldo_llamada
AFTER INSERT ON Llamadas
FOR EACH ROW
EXECUTE FUNCTION consumir_saldo_llamada();

CREATE OR REPLACE FUNCTION actualizar_idconsumible_contrato()
RETURNS TRIGGER AS $$
BEGIN
	UPDATE contratos
	SET id_consumibles = NEW.id_consumibles;
	RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_actualizar_idrecargas_contrato
AFTER INSERT ON Consumiblesxcontrato
FOR EACH ROW
EXECUTE FUNCTION actualizar_idconsumible_contrato();

CREATE OR REPLACE FUNCTION asignar_saldo_inicial()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO Saldo_prepago (numero_abonado, cant_saldo)
    VALUES (NEW.telefono, 0);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_asignar_saldo_inicial
AFTER INSERT ON Clientes
FOR EACH ROW
EXECUTE FUNCTION asignar_saldo_inicial();

--------------------------------------------
CREATE OR REPLACE FUNCTION actualizar_saldo_recarga()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE Saldo_prepago
    SET cant_saldo = cant_saldo + NEW.monto
    WHERE numero_abonado = (
        SELECT telefono
        FROM Clientes
        WHERE id_cliente = (
            SELECT id_cliente
            FROM Contratos
            WHERE id_contrato = NEW.id_contrato
        )
    );
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_actualizar_saldo_recarga
AFTER INSERT ON Recargas
FOR EACH ROW
EXECUTE FUNCTION actualizar_saldo_recarga();

CREATE OR REPLACE FUNCTION consumibles()
RETURNS TRIGGER AS $$
BEGIN
	INSERT INTO consumiblesxcontrato (nombre_plan, minutos, gigas, mensajes, precio_plan_postpago, valor_min_adicional, descripcion, precio_plan_prepago)
	SELECT p.nombre_plan, p.minutos, p.gigas, p.mensajes, p.precio_plan_postpago, p.valor_min_adicional, p.descripcion, p.precio_plan_prepago
	FROM planes p
	WHERE p.id_plan = NEW.id_plan;
	RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_consumibles
AFTER INSERT ON contratos
FOR EACH ROW
EXECUTE FUNCTION consumibles();
